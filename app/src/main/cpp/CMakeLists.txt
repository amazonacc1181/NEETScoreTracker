cmake_minimum_required(VERSION 3.22.1)
project(neetllm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(llama_cpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG master GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(llama_cpp)
add_library(neetllm SHARED neetllm.cpp)
target_link_libraries(neetllm PRIVATE llama android log)
target_compile_options(neetllm PRIVATE -O3 -fvisibility=hidden)
